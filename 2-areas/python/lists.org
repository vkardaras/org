* List Methods

- ~list.append()~
- ~list.insert()~

#+begin_src python
  countries = ["USA", "Canada", "India"]
  # add an item to the end of the list
  countries.append("Spain")
  # insert an item to a specific position of the list
  countries.insert(2, "Italy")
  print(countries)
#+end_src

#+RESULTS:
: ['USA', 'Canada', 'Italy', 'India', 'Spain']

** Swap values inside a list

#+begin_src python
  countries = ["USA", "Canada", "India"]
  temp = countries[0]
  countries[0] = countries[1]
  countries[1] = temp
  print(countries)
#+end_src

#+RESULTS:
: ['Canada', 'USA', 'India']

Easier way to swap values

#+begin_src python
  countries = ["USA", "Canada", "India"]
  countries[0], countries[1] = countries[1], countries[0]
  print(countries)
#+end_src

#+RESULTS:
: ['Canada', 'USA', 'India']

* Iterating Lists

#+begin_src python
  ages = [56,72,24,46]
  total = 0
  for age in ages:
    total += age
  average = total / len(ages)
  print(average)
#+end_src

#+RESULTS:
: 49.5

* Slicing Lists

~list[start:end]~

#+begin_src python
  letters = ["A","B","C","D","E"]
  firstTwo = letters[0:2]
  print(firstTwo)
  print(letters[1:])
  print(letters[:3])
  print(letters[1:-1])
  print(letters[:])
  # print every second element
  print(letters[::2])
  # reverse a list
  print(letters[::-1])
  # delete from 1 until 3 (3 not included)
  del letters[1:3]
  print(letters)
  # delete all elements in an array
  del letters[:]
#+end_src

#+RESULTS:
: ['A', 'B']
: ['B', 'C', 'D', 'E']
: ['A', 'B', 'C']
: ['B', 'C', 'D']
: ['A', 'B', 'C', 'D', 'E']
: ['A', 'C', 'E']
: ['E', 'D', 'C', 'B', 'A']
: ['A', 'D', 'E']
