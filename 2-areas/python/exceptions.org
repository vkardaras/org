*Handled Exceptions*

#+begin_src python
  try:
    name = 'Lydia'
    print('My name is ' + naem)
  except:
    print('Something went wrong')

  print('All done!')
#+end_src

#+RESULTS:
: Something went wrong
: All done!

** Create Exceptions for specific type Errors

#+begin_src python
  try:
    x = int(input("Enter a number: "))
    y = 1 / x
    print(y)
  except ZeroDivisionError:
    print("You cannot devide by zero.")
  except ValueError:
    print("Please enter an integer:")
  except:
    print("Something else went wrong")

  print("All done!")
#+end_src

*Specify more than one exception in =except= branch*

#+begin_src python
  try:
    x = int(input("Enter a number: "))
    y = 1 / x
    print(y)
  except (ZeroDivisionError, ValueError):
    print("Invalid input value")
  except ArithmeticError:
    print("Calculation failed.")
  except:
    print("Something else went wrong")

  print("All done!")
#+end_src

** Handle Exception inside a function

#+begin_src python
  def calculate_use_input():
    try:
      x = int(input("Enter a number:"))
      y = 1 / x
      print(y)
    except ZeroDivisionError:
      print("You cannot divide by zero.")
    except:
      print("Something else went wrong")

    return None

  calculate_use_input()

** Manually raise an exception

#+begin_src python

    print("Something else went wrong")
